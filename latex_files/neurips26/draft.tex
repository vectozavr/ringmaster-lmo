\section{Problem Setup}
% 
% 
We study the nonconvex stochastic optimization problem
\begin{equation}\label{eq:problem}
\minimize_{x \in \R^d} 
\left\{ f(x) \coloneqq \ExpSub{\xi \sim \cD}{f(x;\xi)} \right\} ,
\end{equation}
where $x \in \R^d$ is the model parameter and $\xi \sim \cD$ is a random data sample.

We consider a distributed setting with $n$ workers.
Each worker $i$ has access to the same data distribution~$\cD$.
To model heterogeneous worker speeds, we use the fixed computation model of \citet{mishchenko2022asynchronous}.
% 
\begin{assumption}[Fixed Computation Model]
  \label{assump:fixed_time}
  For each worker $i \in [n]$, computing a stochastic gradient $\nabla f(x;\xi)$ takes $\tau_i > 0$ seconds.
  Without loss of generality, we assume $\tau_1 \le \tau_2 \le \cdots \le \tau_n$.
\end{assumption}
% 
We also assume zero communication delay between the workers and the server.
This assumption is standard in time-complexity analyses of distributed optimization
\citep{mishchenko2022asynchronous, koloskova2022sharper, tyurin2023optimal, maranjyan2025ringmaster}.

We introduce the following notation: linear minimization oracle
$$
  \textrm{lmo}(y) \eqdef \argmin_{x~:~\|x\| \le 1} \inp{y}{x} .
$$
% 
\begin{definition}
  \label{def:gen_smooth}
  Let $f:\R^d\rightarrow\R$ be differentiable.
  Then $f$ is symmetric $(L_0, L_1)$-smooth if there exist $L_0,L_1 \ge 0$ such that for any $x, y \in \R^d$
  \begin{equation}
      \label{eq:gen_smooth}
      \norm{\nabla f(x) - \nabla f(y)}_* \le \sup_{\theta \in [0,1]} \( L_0 + L_1 \norm{\nabla f(\theta x +(1-\theta)y)}_* \) \norm{x-y}.
  \end{equation}
\end{definition}
% 
% 
\section{Algorithm}
% 
% 
% 
\begin{algorithm}[H]
  \caption{LMO \ringmaster (without calculation stops)}
  \label{alg:ringmaster_lmo}
  \begin{algorithmic}[1]
      \STATE \textbf{Input:} points $x^0, m^0 \in \R^d$, stepsize $\eta_k > 0,$ delay threshold $R_k \in \N$, momentum parameter $0<\alpha_k \le 1$.
      \STATE Set $k = 0$
      \STATE Workers start computing stochastic gradients at $x^0$
      \WHILE{not terminated}
          \STATE Gradient $g^k = \nabla f\big(x^{k-\delta^k}; \xi^{k-\delta^k}_{i}\big)$ arrives from worker $i$
          \highlightcolor
          \IF{$\delta^k < R_k$}
              \color{black}
              \STATE Update the momentum: $m^{k+1} = \alpha_k g^k + (1-\alpha_k)m^{k} $
              \STATE Update the model:
                  $x^{k+1} = x^{k} + \eta_k\textrm{lmo}(m^{k+1})$
              \STATE Worker $i$ begins calculating 
                  $\nabla f\big(x^{k+1}; \xi^{k+1}_{i}\big)$
              \STATE Update the iteration number $k = k + 1$
          \ELSE
              \STATE Ignore the outdated gradient 
                  $\nabla f\big(x^{k-\delta^k}; \xi^{k-\delta^k}_{i}\big)$
              \STATE Worker $i$ begins calculating 
                  $\nabla f\big(x^{k}; \xi^{k}_{i}\big)$
          \ENDIF
      \ENDWHILE
  \end{algorithmic}
\end{algorithm}
% 
% 
\begin{algorithm}[H]
  \caption{\ringmaster (with calculation stops)}
  \label{alg:ringmasternewstops}
  \begin{algorithmic}[1]
      \STATE \textbf{Input:} points $x^0, m^0 \in \R^d$, stepsize $\eta_k > 0,$ delay threshold $R_k \in \N$, momentum parameter $0<\alpha_k \le 1$.
      \STATE Set $k = 0$
      \STATE Workers start computing stochastic gradients at $x^0$

          \STATE Worker $i$ begins calculating 
              $\nabla f\big(x^{k+1}; \xi^{k+1}_{i}\big)$
          \STATE Update the iteration number $k = k + 1$
      \ENDWHILE
  \end{algorithmic}
\end{algorithm}

\section{Theoretical Analysis}
% 
\subsection{Iteration Complexity}
% 
\begin{theorem}[Iteration Complexity; Proof in \Cref{proof:iteration_complexity}]\label{thm:iteration_complexity}
  Assume $f$ is $L_0$-smooth, that is, $L_1 = 0$.
  Take $\alpha_0 =1$, $\alpha_k = \nicefrac{1}{\sqrt{k}}$, $R_k = \left\lfloor\nicefrac{1}{\alpha_k}\right\rfloor$, and $\eta_k = \nicefrac{\eta}{(k+1)^{\nicefrac{3}{4}}}$ for $k \ge 1$.
  Then, after $K$ iterations, the iterates satisfy
  \begin{equation*}
      \min_{k \in \{0,\dots,K-1\}} \Exp{\norm{\nabla f(x^k)}_*}
      =
      \mathcal{O}\left(
          \frac{\Psi \log K}{K^{1/4}}
      \right).
  \end{equation*}
  where
  \begin{equation*}
      \Psi \coloneqq \nicefrac{\Delta_0}{\eta} + \rho \sigma + L_0\eta .
  \end{equation*}
  Consequently, the number of iterations needed to reach an $\varepsilon$-stationary point satisfies
  \begin{equation*}
      K_\varepsilon
      = \cO\left(\frac{\Psi^4}{\varepsilon^4} \( \log \frac{1}{\varepsilon} \)^4\right).
   \end{equation*}
\end{theorem}

\subsection{Time Complexity}
% 
We now analyze the wall-clock time complexity of \Cref{alg:ringmaster_lmo}.
The following lemma bounds the wall-clock time needed to complete $K$ iterations under the square-root threshold schedule.
% 
\begin{lemma}[Time Complexity for Square-Root Threshold; Proof in \Cref{proof:sqrt_threshold}]\label{lemma:time_sqrt}
  Consider \Cref{alg:ringmaster_lmo} with delay threshold $R_k = \lfloor \sqrt{k+1} \rfloor$ for $k \ge 0$.
  Under \Cref{assump:fixed_time}, the wall-clock time needed to complete the first $K$ iterations satisfies
  \begin{equation*}
      T(K) = \mathcal{O}\left( \sqrt{K} \min_{m \in [n]} H_m \left( 1 + \frac{\sqrt{K}}{m} \right) \right).
  \end{equation*}
  where
  \begin{equation*}
      H_m \coloneqq \left(\frac{1}{m}\sum_{i=1}^{m}\frac{1}{\tau_i}\right)^{-1}
  \end{equation*}
  is the harmonic mean of the first $m$ computation times $\tau_1,\dots,\tau_m$.
\end{lemma}

% 
\begin{proof}[Proof sketch]
  Define $\mathcal{K}_r \coloneqq \{k \mid \lfloor R_k \rfloor = r\}$, namely the set of iterations for which the threshold equals $r$.
  The schedule $R_k = \lfloor \sqrt{k+1} \rfloor$ partitions the $K$ iterations into these threshold-constant blocks.
  For each block $\mathcal{K}_r$, we apply \Cref{lem:time_R}, which is exactly the fixed-threshold lemma of \citet{maranjyan2025ringmaster} restated in the appendix, and then sum over all values $r \le \lfloor \sqrt{K} \rfloor$.
  Since $|\mathcal{K}_r| = 2r+1 = \mathcal{O}(r)$, each block contributes only a constant number of fixed-threshold chunks, which yields the claim.
\end{proof}
% 
Combining \Cref{thm:iteration_complexity} with \Cref{lemma:time_sqrt} yields the total wall-clock complexity for reaching an $\varepsilon$-stationary point.
% 
\begin{theorem}[Total Wall-Clock Complexity]\label{thm:time_complexity}
  Under the assumptions and parameter choices of \Cref{thm:iteration_complexity}, let $K_\varepsilon$ be the number of iterations needed to reach an $\varepsilon$-stationary point.
  Then, under \Cref{assump:fixed_time}, the corresponding wall-clock time
  $T_\varepsilon \coloneqq T(K_\varepsilon)$ satisfies
  \begin{equation*}
      T_\varepsilon
      =
      \widetilde{\mathcal{O}}\left(
          \frac{\Psi^2}{\varepsilon^2}
          \min_{m \in [n]} H_m \left(
              1 + \frac{\Psi^2}{m\varepsilon^2}
          \right)
      \right) ,
  \end{equation*}
  where $H_m$ is the harmonic mean of the first $m$ computation times $\tau_1,\dots,\tau_m$, $\Psi \coloneqq \nicefrac{\Delta_0}{\eta} + \rho \sigma + L_0\eta$, and $\widetilde{\mathcal{O}}(\cdot)$ hides logarithmic factors.
\end{theorem}

\section{Auxiliary Lemmas}
% 
We assume that the minimized function is symmetric $(L_0,L_1)$-smooth. To prove convergence theorem we state auxiliary lemma with several useful properties. 
\begin{lemma}[\cite{chen2023generalized}]
    \label{lem:gen_smooth}
    Let $f$ be symmetric $(L_0, L_1)$-smooth. Then for any $x,y\in \R^d$ it holds 
    \begin{gather*}
        f(x) \le f(y) + \inp{\nabla f(y)}{x-y} +\frac{1}{2}\left(L_0 + L_1\norm{\nabla f(y)}_*\right)\exp\left(L_1\norm{x-y}\right)\norm{x-y}^2, \\
        \norm{\nabla f(x) - \nabla f(y)}_* \le \left(L_0 + L_1\norm{\nabla f(y)}_*\right)\exp\left(L_1\norm{x-y}\right)\norm{x-y} .
    \end{gather*}
\end{lemma}
Next lemma is a decent lemma for update  $x^{k+1} = x^{k} + \eta_k\textrm{lmo}(m^{k+1})$. 
\begin{lemma}[\cite{khirirat2025better}]
    \label{lem:descent_lemma}
    Let $f$ be symmetric $(L_0, L_1)$-smooth. Then iterates of Algorithm~\ref{alg:ringmaster_lmo} satisfy 
    \begin{equation}
        \sum_{k=0}^{K-1}\eta_k\phi_k\norm{\nabla f(x^k)}_* \leq \Delta_0 + 2\sum^{K-1}_{k=0}\eta_k\norm{\hat{e}_{k+1}}_* +\frac{L_0}{2}\sum^{K-1}_{k=0}\exp\left(L_1\eta_k\right)\eta_k^2 ~,
    \end{equation}
    where we define error term as $\hat{e}_{k+1} \eqdef m^{k+1} - \nabla f(x^k)$ and constant $\phi_k  = 1-\frac{L_1\eta_k}{2}\exp(L_1\eta_k)$. 
\end{lemma}
\begin{proof}
    By Lemma~\ref{lem:gen_smooth}, we have 
    \begin{align*}
        f(x^{k+1})
        &\le f(x^k) + \inp{\nabla f(x^k)}{x^{k+1}-x^k}\\
        &\quad +\frac{1}{2}\left(L_0 + L_1\norm{\nabla f(x^k)}_*\right)\exp\left(L_1\norm{x^{k+1}-x^k}\right)\norm{x^{k+1}-x^k}^2\\
        &\le f(x^k) + \eta_k\inp{\nabla f(x^k)}{\textrm{lmo}(m^{k+1})} +\frac{1}{2}\left(L_0 + L_1\norm{\nabla f(x^k)}_*\right)\exp\left(L_1\eta_k\right)\eta_k^2\\
        &= f(x^k) + \eta_k\inp{\nabla f(x^k) - m^{k+1}}{\textrm{lmo}(m^{k+1})} + \eta_k\inp{m^{k+1}}{\textrm{lmo}(m^{k+1})} \\
        &\quad +\frac{1}{2}\left(L_0 + L_1\norm{\nabla f(x^k)}_*\right)\exp\left(L_1\eta_k\right)\eta_k^2\\
        &\le f(x^k) + \eta_k\norm{\nabla f(x^k) - m^{k+1}}_* - \eta_k\norm{m^{k+1}}_*\\
        &\quad + \frac{1}{2}\left(L_0 + L_1\norm{\nabla f(x^k)}_*\right)\exp\left(L_1\eta_k\right)\eta_k^2\\
        &\le f(x^k) + 2\eta_k\norm{\nabla f(x^k) - m^{k+1}}_* - \eta_k\norm{\nabla f(x^k)}_*\\
        &\quad + \frac{1}{2}\left(L_0 + L_1\norm{\nabla f(x^k)}_*\right)\exp\left(L_1\eta_k\right)\eta_k^2 ~.
    \end{align*}
    For the second inequality we used the fact that $\|\mathrm{lmo}(\cdot)\| \le 1$, then for the 3rd inequality we used Holder inequality and for the last step we used triangle inequality.
\end{proof}
%  
% 
We define
$$
\hat{e}_{k+1} \eqdef m^{k+1} - \nabla f(x^k) ~.
$$
\begin{lemma}
\label{lem:error_bound}
    Let $f$ be symmetric $(L_0, L_1)$-smooth.
    If stepsize $\eta_k$ is non-increasing and momentum parameter $\alpha_k$ defined as $\alpha_0 =1$ and $\alpha_k = \frac{1}{\sqrt{k}}$ for $k \ge 1$, then  the iterates of Algorithm~\ref{alg:ringmaster_lmo} satisfy 
    \begin{align*}
        \Exp{\norm{\hat{e}_{k+1}}_*}
        &\le \sum_{t=1}^k \prod_{j=t+1}^{k}(1-\alpha_j) \left(L_0 + L_1\Exp{\norm{\nabla f(x^t)}_*}\right)\exp(L_1\eta_{t-1})\eta_{t-1}\\
            &\quad + \sum_{t=1}^k \prod_{j=t+1}^{k}(1-\alpha_j) \alpha_t (L_0 + L_1 \Exp{\norm{\nabla f(x^t)}_*})\exp\left(L_1 R_t\eta_{t-\delta_t}\right)R_t\eta_{t-\delta_t}\\
                &\quad + {\color{red} \rho} \sigma\sqrt{\sum_{t=1}^k \prod_{j=t+1}^{k}(1-\alpha_j)^2 \alpha_t^2} ~.
    \end{align*}
\end{lemma}
\begin{proof}
Using momentum update, we derive
\begin{eqnarray*}
    \hat{e}_{k+1}
    &=& m^{k+1} - \nabla f(x^k) \\
    &=& (1-\alpha_k)m^k +\alpha_k g^k - \nabla f(x^k) \\
    &=& (1-\alpha_k)(m^k- \nabla f(x^{k-1})) + (1-\alpha_k)(\nabla f(x^{k-1}) - \nabla f(x^k)) +\alpha_k g^k - \alpha_k\nabla f(x^k) \\
    &=& (1-\alpha_k)(m^k- \nabla f(x^{k-1})) + (1-\alpha_k)(\nabla f(x^{k-1}) - \nabla f(x^k)) \\
    && +\alpha_k (\nabla f(x^{k-\delta_k}) - \nabla f(x^k))  +\alpha_k(g^k - \nabla f(x^{k-\delta_k}) ~.
\end{eqnarray*}
% 
Denoting $D_k \eqdef   \nabla f(x^{k-1}) - \nabla f(x^k)$, $S_k \eqdef \nabla f(x^{k-\delta_k}) - \nabla f(x^k)$, and $e_k = g^k - \nabla f(x^{k-\delta_k})$, we obtain
\begin{eqnarray*}
    \hat{e}_{k+1}
    &=& (1-\alpha_k) \hat{e}_{k} + (1-\alpha_k) D_k + \alpha_k S_k +\alpha_k e_k \\
    &=&\prod_{t = 0}^{k}(1-\alpha_t) \hat{e}_{0} + \sum_{t=0}^k \prod_{j=t+1}^{k}(1-\alpha_j)(1-\alpha_t) D_t \\
    && + \sum_{t=0}^k \prod_{j=t+1}^{k}(1-\alpha_j) \alpha_t S_t + \sum_{t=0}^k \prod_{j=t+1}^{k}(1-\alpha_j) \alpha_t e_t ~.
\end{eqnarray*}
Setting $\alpha_k$ as $\alpha_0 = \alpha_1 =1$ and $\alpha_k = \frac{1}{\sqrt{k}}$ for $k \ge 1$, we have
% 
\begin{eqnarray*}
    \hat{e}_{k+1} &=& \sum_{t=0}^k \prod_{j=t+1}^{k}(1-\alpha_j)(1-\alpha_t) D_t  + \sum_{t=0}^k \prod_{j=t+1}^{k}(1-\alpha_j) \alpha_t S_t + \sum_{t=0}^k \prod_{j=t+1}^{k}(1-\alpha_j) \alpha_t e_t\\
    &=& \underbrace{\sum_{t=1}^k \prod_{j=t+1}^{k}(1-\alpha_j)(1-\alpha_t) D_t}_{\eqdef\cT_1} + \underbrace{\sum_{t=1}^k \prod_{j=t+1}^{k}(1-\alpha_j) \alpha_t S_t}_{\eqdef \cT_2} + \underbrace{\sum_{t=1}^k \prod_{j=t+1}^{k}(1-\alpha_j) \alpha_t e_t}_{\eqdef \cT_3}.
\end{eqnarray*}
% 
To continue the proof, we need to bound the norm of terms $\cT_1$, $\cT_2$ and $\cT_3$. By triangle inequality, we get 
\begin{eqnarray*}
    \norm{\cT_1}_* &\le& \sum_{t=1}^k \prod_{j=t+1}^{k}(1-\alpha_j)(1-\alpha_t) \norm{D_t}_*\\
    &=& \sum_{t=1}^k \prod_{j=t+1}^{k}(1-\alpha_j) \norm{ \nabla f(x^{t-1}) - \nabla f(x^t)}_*\\
    &\le& \sum_{t=1}^k \prod_{j=t+1}^{k}(1-\alpha_j) \left(L_0 + L_1\norm{\nabla f(x^t)}_*\right)\exp(L_1\eta_{t-1})\eta_{t-1} ~.
\end{eqnarray*}
We used \Cref{lem:gen_smooth} in the last inequality.
% 
Bounding $\norm{\cT_2}_*$, we obtain 
\begin{eqnarray*}
    \norm{\cT_2}_* &\le& \sum_{t=1}^k \prod_{j=t+1}^{k}(1-\alpha_j) \alpha_t \norm{S_t}_*\\
    &=& \sum_{t=1}^k \prod_{j=t+1}^{k}(1-\alpha_j) \alpha_t \norm{\nabla f(x^{t-\delta_t}) - \nabla f(x^t)}_*\\
    &\le& \sum_{t=1}^k \prod_{j=t+1}^{k}(1-\alpha_j) \alpha_t \(L_0 + L_1 \norm{\nabla f(x^t)}_*\)\exp\left(L_1 \norm{x^t - x^{t-\delta_t}}\right)\norm{x^t - x^{t-\delta_t}}\\
    &\le& \sum_{t=1}^k \prod_{j=t+1}^{k}(1-\alpha_j) \alpha_t \(L_0 + L_1 \norm{\nabla f(x^t)}_*\)\\

     \end{equation*}

\end{restate-theorem}

    \sum_{t=1}^k \prod_{j=t+1}^{k}(1-\alpha_j)^2 \alpha_t^2 &\le& \sum_{t=1}^k \prod_{j=t+1}^{k}(1-\alpha_j) \alpha_t^2 =  \sum_{t=1}^k \prod_{j=t+1}^{k}(1-j^{-\nicefrac{1}{2}}) t^{-1}\\

\section{Time Complexity Analysis for the Fixed-Time Case}
\arto{name?}
% 
We now study the wall-clock complexity under \Cref{assump:fixed_time}.
For $m \in [n]$, define
\begin{equation*}
    H_m \coloneqq \left(\frac{1}{m}\sum_{i=1}^{m}\frac{1}{\tau_i}\right)^{-1},
\end{equation*}
the harmonic mean of the first $m$ computation times $\tau_1,\dots,\tau_m$.
We start with the case of a fixed delay threshold.
% 
\begin{lemma}[Duration of $R$ updates {\citep[Lemma 4.1]{maranjyan2025ringmaster}}]\label{lem:time_R}
    Let the delay threshold of \Cref{alg:ringmaster_lmo} be fixed as $R_k \equiv R$.
    Under \Cref{assump:fixed_time}, the wall-clock time needed to complete any $R$ consecutive updates is at most
    \begin{equation}\label{eq:time_R}
       t(R) \coloneqq 2 \min_{m \in [n]} H_m \left( 1 + \frac{R}{m} \right).
    \end{equation}
\end{lemma}
% 
As an immediate consequence, if the threshold remains fixed at $R$ for the first $K$ updates, then partitioning these updates into $\lceil K/R \rceil$ consecutive blocks and applying \Cref{lem:time_R} gives the following bound on the wall-clock time needed to complete the first $K$ updates:
\begin{equation*}
    T(K) \le \left\lceil \frac{K}{R} \right\rceil t(R).
\end{equation*}
This is exactly how the time complexity of \ringmaster~\citep{maranjyan2025ringmaster} is obtained.
Our case is slightly more delicate, since the threshold does not remain fixed throughout the run.
To handle this, we extend the same blockwise argument to a non-decreasing threshold sequence.
% 
\begin{lemma}[Time for Non-Decreasing Adaptive Threshold]\label{lem:time_RK}
    Let $R_k$ be a non-decreasing delay threshold.
    For each integer $r \in \{\lfloor R_0 \rfloor, \dots, \lfloor R_{K-1} \rfloor\}$, let
    \begin{equation*}
        \mathcal{K}_r \coloneqq \{k \mid \lfloor R_k \rfloor = r\}
    \end{equation*}
    denote the corresponding block of iterations, and define
    \begin{equation*}
        N_r \coloneqq \left\lceil \frac{|\mathcal{K}_r|}{r} \right\rceil .
    \end{equation*}
    Then the wall-clock time needed by \Cref{alg:ringmaster_lmo} to complete its first $K$ updates satisfies
    \begin{equation*}
        T(K) \le \sum_{r=\lfloor R_0 \rfloor}^{\lfloor R_{K-1} \rfloor} N_r\, t(r) ~.
    \end{equation*}
\end{lemma}
% 
\begin{proof}
    Split the iterations $\{0,1,\dots,K-1\}$ into the blocks $\mathcal{K}_r$.
    Since $R_k$ is non-decreasing, all iterations in each $\mathcal{K}_r$ appear next to each other in time.
    For a fixed $r$, partition $\mathcal{K}_r$ into
    \begin{equation*}
        N_r = \left\lceil \frac{|\mathcal{K}_r|}{r} \right\rceil
    \end{equation*}
    consecutive groups, each containing at most $r$ iterations.
    Throughout each such group, the threshold is equal to $r$.
    If a group contains exactly $r$ iterations, then \Cref{lem:time_R} shows that it is completed within time at most $t(r)$.
    If the last group contains fewer than $r$ iterations, its duration is still at most $t(r)$, since completing fewer than $r$ updates cannot take longer than completing $r$ updates under the same threshold.
    Therefore, the total time spent on all iterations in $\mathcal{K}_r$ is at most $N_r t(r)$.
    Summing this bound over all values of $r$ gives
    \begin{equation*}
        T(K) \le \sum_{r=\lfloor R_0 \rfloor}^{\lfloor R_{K-1} \rfloor} N_r\, t(r) ~.
    \end{equation*}
\end{proof}

\subsection{Proof of \Cref{lemma:time_sqrt}}\label{proof:sqrt_threshold}
% 
We now specialize \Cref{lem:time_RK} to the square-root threshold schedule.
\begin{restate-lemma}{\ref{lemma:time_sqrt}}[Time Complexity for Square-Root Threshold]
    Let the delay threshold be $R_k = \lfloor \sqrt{k+1} \rfloor$ for $k \ge 0$.
    Under \Cref{assump:fixed_time}, the wall-clock time needed to complete $K$ iterations satisfies
    \begin{equation*}
        T(K)
        = \mathcal{O}\left(
            \sqrt{K}\min_{m \in [n]} H_m \left( 1 + \frac{\sqrt{K}}{m} \right)
        \right).
    \end{equation*}
    where
    \begin{equation*}
        H_m \coloneqq \left(\frac{1}{m}\sum_{i=1}^{m}\frac{1}{\tau_i}\right)^{-1}
    \end{equation*}
    is the harmonic mean of the first $m$ computation times $\tau_1,\dots,\tau_m$.
\end{restate-lemma}
% 
\begin{proof}
    Apply \Cref{lem:time_RK} with $R_k = \lfloor \sqrt{k+1} \rfloor$.
    For any integer $r \ge 1$, the condition $\lfloor \sqrt{k+1} \rfloor = r$ is equivalent to
    \begin{equation*}
        r^2 \le k+1 < (r+1)^2.
    \end{equation*}
    Therefore,
    \begin{equation*}
        \mathcal{K}_r = \{r^2-1, r^2, \dots, (r+1)^2-2\}
        \quad \text{and} \quad
        |\mathcal{K}_r| = (r+1)^2-r^2 = 2r+1.
    \end{equation*}
    Hence
    \begin{equation*}
        N_r = \left\lceil \frac{|\mathcal{K}_r|}{r} \right\rceil
        = \left\lceil \frac{2r+1}{r} \right\rceil
        = 3.
    \end{equation*}
    Let $r_{\max} \coloneqq \lfloor \sqrt{K} \rfloor$.
    Using \Cref{lem:time_RK} and \Cref{eq:time_R}, we obtain
    \begin{equation*}
        T(K) \le \sum_{r=1}^{r_{\max}} N_r\, t(r)
        = \sum_{r=1}^{r_{\max}} 3\, t(r)
        = 6 \sum_{r=1}^{r_{\max}} \min_{m \in [n]} H_m \left( 1 + \frac{r}{m} \right) .
    \end{equation*}
    Since $\sum_r \min_m a_{r,m} \le \min_m \sum_r a_{r,m}$, it follows that
    \begin{align*}
        T(K)
        &\le 6 \min_{m \in [n]} 
            H_m \sum_{r=1}^{r_{\max}} \left( 1 + \frac{r}{m} \right)
        \\
        &= 6 \min_{m \in [n]} \left[
            H_m r_{\max}
            + H_m \frac{r_{\max}(r_{\max}+1)}{2m}
        \right].
    \end{align*}
    Since $r_{\max} \le \sqrt{K}$, we conclude that
    \begin{equation*}
        T(K)
        = \mathcal{O}\left( \min_{m \in [n]} H_m \left( \sqrt{K} + \frac{K}{m} \right) \right)
        = \mathcal{O}\left( \sqrt{K} \min_{m \in [n]} H_m \left( 1 + \frac{\sqrt{K}}{m} \right) \right).
    \end{equation*}
    This proves the claim.
\end{proof}

\begin{lemma}[Time for General Adaptive Threshold under the Universal Computation Model]
    \label{lem:time_RK_arbitrary}
    Let $R_k$ be a non-decreasing delay threshold.
    For each integer $r \in \{\lfloor R_0 \rfloor, \dots, \lfloor R_{K-1} \rfloor\}$, let
    $$
        \mathcal{K}_r \coloneqq \{k \mid \lfloor R_k \rfloor = r\}
    $$
    denote the block of iterations where the threshold is constant.
    For every such $r$, partition $\mathcal{K}_r$ into
    $$
        N_r \coloneqq \left\lceil \frac{|\mathcal{K}_r|}{r} \right\rceil
    $$
    consecutive sub-blocks, each containing at most $r$ iterations.

    Let $B \coloneqq \sum_{r=\lfloor R_0 \rfloor}^{\lfloor R_{K-1} \rfloor} N_r$, and enumerate all these sub-blocks in chronological order.
    Denote by $r_j$ the threshold value attached to the $j$-th sub-block, where $j = 1, \dots, B$.
    Define the sequence $\{T_j\}_{j=0}^{B}$ recursively by
    \begin{align*}
        T_0 &\coloneqq 0, \\
        T_j &\coloneqq T_{j-1} + t(r_j; T_{j-1}), \qquad j = 1, \dots, B.
    \end{align*}
    Then the wall-clock time needed to complete the first $K$ iterations of \Cref{alg:ringmaster_lmo} satisfies
    \begin{equation*}
        T(K) \le T_B.
    \end{equation*}
\end{lemma}

\begin{proof}
    Partition the iterations $\{0,1,\dots,K-1\}$ into disjoint blocks $\mathcal{K}_r$ according to the integer value of $\lfloor R_k \rfloor$.
    Since $R_k$ is non-decreasing, all iterations in each $\mathcal{K}_r$ appear next to each other in time.

    Fix some $r$ and consider one of its sub-blocks.
    By construction, this sub-block contains at most $r$ consecutive iterations, and throughout this sub-block the threshold satisfies
    \begin{equation*}
        r \le R_k < r+1.
    \end{equation*}
    Replacing the actual threshold by $r$ can only make the rule more conservative on this sub-block.
    Hence \Cref{lem:time_R_arbitrary} implies that, if this sub-block starts at time $T$, then all its iterations are completed within time at most $t(r;T)$.

    Now enumerate all sub-blocks in chronological order and let $r_j$ be the threshold associated with the $j$-th one.
    Starting from time $T_0 = 0$, the first sub-block is completed by time at most
    \begin{equation*}
        T_1 = T_0 + t(r_1;T_0).
    \end{equation*}
    Repeating the same argument inductively, if the first $j-1$ sub-blocks are completed by time at most $T_{j-1}$, then the $j$-th sub-block is completed by time at most
    \begin{equation*}
        T_j = T_{j-1} + t(r_j;T_{j-1}).
    \end{equation*}
    After all
    \begin{equation*}
        B = \sum_{r=\lfloor R_0 \rfloor}^{\lfloor R_{K-1} \rfloor} \left\lceil \frac{|\mathcal{K}_r|}{r} \right\rceil
    \end{equation*}
    sub-blocks are processed, all first $K$ iterations have been completed. Therefore,
    \begin{equation*}
        T(K) \le T_B.
    \end{equation*}
\end{proof}

This is the analogue of \Cref{lem:time_RK} for the universal computation model.
It reduces the changing-threshold analysis to a sequence of threshold-constant sub-blocks.

\begin{lemma}[Square-Root Threshold under the Universal Computation Model]
    \label{lem:time_sqrt_arbitrary}
    Let $R_k = \lfloor \sqrt{k+1} \rfloor$ for $k \ge 0$.
    Under \Cref{assump:universal_time}, the wall-clock time needed to complete the first $K$ iterations of \Cref{alg:ringmaster_lmo} is bounded by
    \begin{equation*}
        T(K) \le S_{3\lfloor \sqrt{K} \rfloor},
    \end{equation*}
    where the sequence $\{S_j\}_{j=0}^{3\lfloor \sqrt{K} \rfloor}$ is defined recursively as
    \begin{align*}
        S_0 &\coloneqq 0, \\
        S_j &\coloneqq S_{j-1} + t\!\left(\left\lceil \frac{j}{3} \right\rceil; S_{j-1}\right), \qquad j = 1, \dots, 3\lfloor \sqrt{K} \rfloor .
    \end{align*}
\end{lemma}

\begin{proof}
    We apply \Cref{lem:time_RK_arbitrary} with $R_k = \lfloor \sqrt{k+1} \rfloor$.
    For every integer $r \ge 1$, the condition $\lfloor \sqrt{k+1} \rfloor = r$ is equivalent to
    $$
        r^2 \le k+1 < (r+1)^2,
    $$
    hence
    $$
        \mathcal{K}_r = \{r^2-1, r^2, \dots, (r+1)^2-2\},
    $$
    and therefore
    \begin{equation*}
        |\mathcal{K}_r| = (r+1)^2-r^2 = 2r+1.
    \end{equation*}
    Consequently,
    \begin{equation*}
        \left\lceil \frac{|\mathcal{K}_r|}{r} \right\rceil
        =
        \left\lceil \frac{2r+1}{r} \right\rceil
        =
        3.
    \end{equation*}
    Thus each threshold level $r$ contributes exactly three sub-blocks of size at most $r$.

    Let $r_{\max} \coloneqq \lfloor \sqrt{K} \rfloor$.
    Enumerating all sub-blocks in chronological order gives exactly $3r_{\max}$ sub-blocks, with threshold values
    $$
        1,1,1,2,2,2,\dots,r_{\max},r_{\max},r_{\max}.
    $$
    Therefore, the recursion from \Cref{lem:time_RK_arbitrary} becomes
    \begin{align*}
        S_0 &\coloneqq 0, \\
        S_j &\coloneqq S_{j-1} + t\!\left(\left\lceil \frac{j}{3} \right\rceil; S_{j-1}\right), \qquad j = 1, \dots, 3r_{\max},
    \end{align*}
    and the resulting completion time satisfies
    \begin{equation*}
        T(K) \le S_{3r_{\max}} = S_{3\lfloor \sqrt{K} \rfloor}.
    \end{equation*}
\end{proof}

\begin{theorem}[Final Arbitrary-Time Wall-Clock Complexity]
    \label{thm:time_arbitrary}
    Consider \Cref{alg:ringmaster_lmo} with
    $$
        \alpha_0 = 1,\qquad
        \alpha_k = \frac{1}{\sqrt{k}},\qquad
        R_k = \lfloor \sqrt{k+1} \rfloor,
        \qquad
        \eta_k = \frac{\eta}{(k+1)^{3/4}}
    $$
    for $k \ge 1$.
    Let
    \begin{equation*}
        K_{\varepsilon}
        =
        \widetilde{\mathcal{O}}\left(\frac{\Psi^4}{\varepsilon^4}\right)
    \end{equation*}
    be the iteration complexity from \Cref{thm:iteration_complexity}.
    Then, under \Cref{assump:universal_time}, the wall-clock time needed to reach an $\varepsilon$-stationary point satisfies
    \begin{equation*}
        T_{\varepsilon}
        \le
        S_{3\lfloor \sqrt{K_{\varepsilon}} \rfloor}
        =
        S_{\widetilde{\mathcal{O}}(\Psi^2/\varepsilon^2)},
    \end{equation*}
    Here and below, $\widetilde{\mathcal{O}}(\cdot)$ hides logarithmic factors in $1/\varepsilon$.
    The sequence $\{S_j\}$ is defined recursively by
    \begin{align*}
        S_0 &\coloneqq 0, \\
        S_j &\coloneqq S_{j-1} + t\!\left(\left\lceil \frac{j}{3} \right\rceil; S_{j-1}\right).
    \end{align*}
\end{theorem}

\begin{proof}
    By \Cref{thm:iteration_complexity}, the method reaches an $\varepsilon$-stationary point within
    \begin{equation*}
        K_{\varepsilon}
        =
        \widetilde{\mathcal{O}}\left(\frac{\Psi^4}{\varepsilon^4}\right)
    \end{equation*}
    iterations.
    Applying \Cref{lem:time_sqrt_arbitrary} with $K = K_{\varepsilon}$ yields
    \begin{equation*}
        T_{\varepsilon}
        \le
        S_{3\lfloor \sqrt{K_{\varepsilon}} \rfloor}.
    \end{equation*}
    Since $\sqrt{K_{\varepsilon}} = \widetilde{\mathcal{O}}(\Psi^2/\varepsilon^2)$, the claim follows.
\end{proof}

\subsection{Comparison Under Additional Regularity Assumptions}
\label{sec:comparison_under_regular_assumptions}

The comparison above treats the functions $t(R;T)$ as completely arbitrary.
If we are willing to assume that block durations are stable under batching and under shifts in the starting time, then the recursion from \Cref{thm:time_arbitrary} simplifies substantially and the wall-clock behavior of our method becomes very close to that of fixed-threshold \ringmaster.

\begin{assumption}[Bounded Batching Overhead]
    \label{assump:batching_overhead}
    There exists a constant $C_{\rm blk} \ge 1$ such that the following holds.
    For every integer $R \ge 1$, every integer $q \ge 1$, and every starting time $T_0 \ge 0$, define
    \begin{equation*}
        T_{j+1} \coloneqq T_j + t(R;T_j),
        \qquad j = 0,\dots,q-1.
    \end{equation*}
    Then
    \begin{equation*}
        \sum_{j=0}^{q-1} t(R;T_j)
        \le
        C_{\rm blk}\, t(qR;T_0).
    \end{equation*}
\end{assumption}

Intuitively, \Cref{assump:batching_overhead} says that splitting $qR$ updates into $q$ consecutive blocks of size $R$ does not increase the elapsed time by more than a constant factor.
This is reasonable when worker speeds do not change too much over the time window covered by a few consecutive blocks, so that batching several smaller blocks together does not create a dramatic speedup.

\paragraph{Examples where \Cref{assump:batching_overhead} is satisfied under \Cref{assump:universal_time}.}
Recall that
\begin{equation*}
    t(R;T)
    =
    \min \left\{t \ge 0 :
    \sum_{i=1}^{n}
    \left\lfloor
        \frac{1}{4}\int_{T}^{T+t} p_i(s)\,ds
    \right\rfloor
    \ge R
    \right\}.
\end{equation*}
For the examples below, the basic strategy is always the same: we upper bound each small-block duration $t(R;T_j)$ and lower bound the batched duration $t(qR;T_0)$.

A first exact example is a single worker with constant rate $p_1(s)\equiv c$.
Then for any interval of length $t$,
\begin{equation*}
    \left\lfloor
        \frac{1}{4}\int_{T}^{T+t} p_1(s)\,ds
    \right\rfloor
    =
    \left\lfloor \frac{ct}{4} \right\rfloor .
\end{equation*}
Hence the condition
\begin{equation*}
    \left\lfloor \frac{ct}{4} \right\rfloor \ge R
\end{equation*}
is equivalent to
\begin{equation*}
    t \ge \frac{4R}{c},
\end{equation*}
so
\begin{equation*}
    t(R;T)=\frac{4R}{c}
    \qquad
    \text{for all } R \ge 1,\; T \ge 0.
\end{equation*}
Therefore
\begin{equation*}
    \sum_{j=0}^{q-1} t(R;T_j)
    =
    q\,\frac{4R}{c}
    =
    \frac{4qR}{c}
    =
    t(qR;T_0),
\end{equation*}
and one may take $C_{\rm blk}=1$.

The same idea works for multiple workers with constant rates $p_i(s)\equiv c_i$.
Write
\begin{equation*}
    c_{\rm tot}\coloneqq \sum_{i=1}^{n} c_i.
\end{equation*}
For an interval of length $t$, we have
\begin{equation*}
    \sum_{i=1}^{n}\left\lfloor \frac{c_i t}{4} \right\rfloor
    \le
    \frac{c_{\rm tot} t}{4},
\end{equation*}
and also
\begin{equation*}
    \sum_{i=1}^{n}\left\lfloor \frac{c_i t}{4} \right\rfloor
    \ge
    \sum_{i=1}^{n}\left(\frac{c_i t}{4}-1\right)
    =
    \frac{c_{\rm tot} t}{4}-n.
\end{equation*}
The upper bound says that one cannot finish $qR$ updates before time $4qR/c_{\rm tot}$, so
\begin{equation*}
    t(qR;T_0)\ge \frac{4qR}{c_{\rm tot}}.
\end{equation*}
The lower bound says that by time $4(R+n)/c_{\rm tot}$ one has certainly finished at least $R$ updates, so
\begin{equation*}
    t(R;T)=t(R;0)\le \frac{4(R+n)}{c_{\rm tot}}.
\end{equation*}
Since the rate is constant, every small block has the same duration bound, and therefore
\begin{equation*}
    \sum_{j=0}^{q-1} t(R;T_j)
    =
    q\, t(R;0)
    \le
    q\,\frac{4(R+n)}{c_{\rm tot}}
    \le
    (n+1)\,\frac{4qR}{c_{\rm tot}}
    \le
    (n+1)\, t(qR;T_0).
\end{equation*}
Thus one may take $C_{\rm blk}=n+1$.
This models a dedicated cluster with reserved GPUs or a tightly controlled on-premise system, where the effective worker speeds stay essentially constant during the run.

The bounded-envelope condition from \Cref{assump:start_time_distortion} also works here.
Assume that there exist numbers $q_-,q_+>0$ and nominal times $\tau_i>0$ such that
\begin{equation*}
    \frac{q_-}{\tau_i}
    \le
    p_i(s)
    \le
    \frac{q_+}{\tau_i},
    \qquad i \in [n],\; s \ge 0.
\end{equation*}
Write
\begin{equation*}
    S \coloneqq \sum_{i=1}^{n}\frac{1}{\tau_i}.
\end{equation*}
Then for every interval $[T,T+t]$,
\begin{equation*}
    \sum_{i=1}^{n}
    \left\lfloor
        \frac{1}{4}\int_{T}^{T+t} p_i(s)\,ds
    \right\rfloor
    \le
    \frac{q_+ S t}{4},
\end{equation*}
because we may simply drop the floors and use the upper envelope.
Likewise,
\begin{equation*}
    \sum_{i=1}^{n}
    \left\lfloor
        \frac{1}{4}\int_{T}^{T+t} p_i(s)\,ds
    \right\rfloor
    \ge
    \sum_{i=1}^{n}
    \left(
        \frac{1}{4}\int_{T}^{T+t} p_i(s)\,ds - 1
    \right)
    \ge
    \frac{q_- S t}{4} - n.
\end{equation*}
Exactly as above, these inequalities imply
\begin{equation*}
    t(qR;T_0)\ge \frac{4qR}{q_+ S}
    \qquad\text{and}\qquad
    t(R;T)\le \frac{4(R+n)}{q_- S}
\end{equation*}
for every $T \ge 0$.
Therefore
\begin{equation*}
    \sum_{j=0}^{q-1} t(R;T_j)
    \le
    q\,\frac{4(R+n)}{q_- S}
    \le
    \frac{q_+}{q_-}(n+1)\,\frac{4qR}{q_+ S}
    \le
    \frac{q_+}{q_-}(n+1)\, t(qR;T_0).
\end{equation*}
So this envelope condition implies \Cref{assump:batching_overhead} with
\begin{equation*}
    C_{\rm blk} = \frac{q_+}{q_-}(n+1).
\end{equation*}
This is a useful sufficient condition in realistic systems where each worker has a nominal speed $1/\tau_i$, but the whole cluster is modulated by a global factor that stays inside a bounded range.
Examples include mild co-tenancy, bounded thermal throttling, or scheduled I/O and networking slowdowns that affect all workers comparably.

A slightly more general sufficient condition is to bound only the aggregate throughput.
If there exist constants $P_-,P_+>0$ such that
\begin{equation*}
    P_-
    \le
    \sum_{i=1}^{n} p_i(s)
    \le
    P_+
    \qquad\text{for all } s \ge 0,
\end{equation*}
then the same argument gives
\begin{equation*}
    t(qR;T_0)\ge \frac{4qR}{P_+}
    \qquad\text{and}\qquad
    t(R;T)\le \frac{4(R+n)}{P_-},
\end{equation*}
hence
\begin{equation*}
    \sum_{j=0}^{q-1} t(R;T_j)
    \le
    \frac{P_+}{P_-}(n+1)\, t(qR;T_0).
\end{equation*}
This covers systems where individual worker speeds may fluctuate, but the total amount of compute available to the job stays within a fixed envelope.

The previous sufficient conditions suggest the practical regime in which \Cref{assump:batching_overhead} should be expected to hold.
If the total throughput of the system does not change too much over the time needed to process the $q$ consecutive sub-blocks, then the split execution and the batched execution see essentially the same pool of compute, and the difference between them comes only from floor effects and from where the block boundaries are placed.
This is what happens, approximately, in cloud jobs running inside a stable quota window, in overnight training runs with only mild background variation, or in managed services where throughput changes are infrequent and bounded.

\paragraph{Examples where \Cref{assump:batching_overhead} may fail.}
The key failure mode is repeated loss of near-complete work at block boundaries.
If a worker contributes almost one full update in each block, but not quite enough to cross the next integer, then each small block discards that partial progress because of the floor in the definition of $t(R;T)$.
By contrast, a single batched block can accumulate those nearly complete contributions over many epochs before the floor is applied.

A concrete construction is the following.
Take two workers and threshold $R=1$.
Partition time into epochs $[E_k,E_{k+1})$, and choose the rates so that over epoch $k$,
\begin{equation*}
    \frac{1}{4}\int_{E_k}^{E_{k+1}} p_1(s)\,ds = 1,
    \qquad
    \frac{1}{4}\int_{E_k}^{E_{k+1}} p_2(s)\,ds = 1-\varepsilon
\end{equation*}
for some fixed $\varepsilon \in (0,1)$.
One can realize this with piecewise-constant rates, for example by spreading most of the mass over a long interval of length $L_k$ and using a short burst near the end of the epoch.

Now start a block of size $R=1$ at time $E_k$.
By the end of the epoch, the total number of completed updates is
\begin{equation*}
    \left\lfloor 1 \right\rfloor
    +
    \left\lfloor 1-\varepsilon \right\rfloor
    =
    1.
\end{equation*}
So one small block needs essentially one whole epoch.
If we repeat this $q$ times, starting each new block at the next epoch boundary, then the second worker's contribution $1-\varepsilon$ is lost every time.
Hence
\begin{equation*}
    \sum_{j=0}^{q-1} t(1;T_j)
    \asymp
    \sum_{k=1}^{q} L_k.
\end{equation*}

Now compare this with one batched block of size $q$ started at $E_1$.
After $m$ epochs, worker $1$ has contributed exactly $m$ completed updates, while worker $2$ has contributed $(1-\varepsilon)m$ updates before flooring.
Therefore the total number of completed updates available to the batched block is at least
\begin{equation*}
    \left\lfloor m \right\rfloor
    +
    \left\lfloor (1-\varepsilon)m \right\rfloor
    \ge
    (2-\varepsilon)m - 1.
\end{equation*}
Thus it suffices to take
\begin{equation*}
    m \asymp \frac{q}{2-\varepsilon},
\end{equation*}
so that
\begin{equation*}
    t(q;E_1)
    \asymp
    \sum_{k=1}^{\lceil q/(2-\varepsilon)\rceil} L_k.
\end{equation*}
If the epoch lengths grow rapidly, say $L_k = 2^k$, then
\begin{equation*}
    \frac{\sum_{j=0}^{q-1} t(1;T_j)}{t(q;E_1)}
    \asymp
    \frac{\sum_{k=1}^{q} 2^k}
         {\sum_{k=1}^{\lceil q/(2-\varepsilon)\rceil} 2^k}
\end{equation*}
diverges as $q \to \infty$.
Hence no uniform constant $C_{\rm blk}$ can satisfy \Cref{assump:batching_overhead}.

At a practical level, this failure mode corresponds to highly bursty systems in which some workers repeatedly provide almost-enough progress but block boundaries prevent this progress from being reused.
Examples include sharply time-varying spot or volunteer pools, or systems with strong synchronization effects in which useful work arrives in short bursts separated by long quiet periods.

\begin{assumption}[Bounded Start-Time Distortion]
    \label{assump:start_time_distortion}
    There exists a constant $C_{\rm time} \ge 1$ such that
    \begin{equation*}
        t(R;T)
        \le
        C_{\rm time}\, t(R;0)
    \end{equation*}
    for every integer $R \ge 1$ and every $T \ge 0$.
\end{assumption}

Intuitively, \Cref{assump:start_time_distortion} says that the main determinant of the block duration is its size $R$, while the precise starting time affects this duration only up to a constant factor.
This may hold in practice when the system runs in a roughly stationary regime, or more generally when the worker speeds stay within a bounded envelope over time.
For example, the stronger condition
\begin{equation*}
    \frac{q_-}{\tau_i}
    \le
    p_i(s)
    \le
    \frac{q_+}{\tau_i},
    \qquad i \in [n], \quad s \ge 0,
\end{equation*}
implies \Cref{assump:start_time_distortion} with $C_{\rm time} = q_+/q_-$.

\paragraph{Examples under \Cref{assump:universal_time}.}
The cleanest example is again the constant-rate model $p_i(s)\equiv c_i$, in which case
\begin{equation*}
    t(R;T)=t(R;0)
\end{equation*}
for all $R \ge 1$ and $T \ge 0$, so one may take $C_{\rm time}=1$.
More generally, if all worker rates share the same bounded time modulation,
\begin{equation*}
    p_i(s)=\frac{a(s)}{\tau_i},
    \qquad
    q_- \le a(s) \le q_+,
\end{equation*}
then every interval of length $t$ delivers between $q_-$ and $q_+$ times the amount of work produced by the fixed computation model, so the duration of a block of size $R$ can change with its starting time only by the factor $q_+/q_-$.
This covers realistic systems with bounded co-tenancy or mild thermal throttling, where the whole cluster speeds up or slows down over time but only within a controlled envelope.
Another concrete example is a scheduled training window on edge devices or lab machines, where
\begin{equation*}
    p_i(s)=\frac{a(s)}{\tau_i}
\end{equation*}
and $a(s)$ is lower during a short busy period and higher during a mostly idle period, while remaining bounded away from both $0$ and $\infty$.
In that case, starting a block a bit earlier or later changes its duration only by a constant factor.

The next lemma shows that \Cref{assump:start_time_distortion} already removes the recursive dependence on the intermediate starting times.
\begin{lemma}[Simplified Formula Under Bounded Start-Time Distortion]
    \label{lem:simplified_formula_start_time}
    Under \Cref{assump:start_time_distortion}, the square-root schedule satisfies
    \begin{equation*}
        T(K)
        \le
        3 C_{\rm time}\sum_{r=1}^{\lfloor \sqrt{K} \rfloor} t(r;0).
    \end{equation*}
    Consequently,
    \begin{equation*}
        T_{\varepsilon}
        \le
        3 C_{\rm time}\sum_{r=1}^{\lfloor \sqrt{K_{\varepsilon}} \rfloor} t(r;0).
    \end{equation*}
\end{lemma}

\begin{proof}
    By \Cref{lem:time_sqrt_arbitrary},
    \begin{equation*}
        T(K)
        \le
        S_{3\lfloor \sqrt{K} \rfloor},
    \end{equation*}
    where
    \begin{equation*}
        S_j
        =
        S_{j-1}
        +
        t\!\left(\left\lceil \frac{j}{3} \right\rceil; S_{j-1}\right).
    \end{equation*}
    Under \Cref{assump:start_time_distortion}, each increment is bounded by
    \begin{equation*}
        t\!\left(\left\lceil \frac{j}{3} \right\rceil; S_{j-1}\right)
        \le
        C_{\rm time}\,
        t\!\left(\left\lceil \frac{j}{3} \right\rceil; 0\right).
    \end{equation*}
    Summing over $j = 1,\dots,3\lfloor \sqrt{K} \rfloor$ gives
    \begin{equation*}
        T(K)
        \le
        3 C_{\rm time}\sum_{r=1}^{\lfloor \sqrt{K} \rfloor} t(r;0).
    \end{equation*}
    The bound for $T_{\varepsilon}$ follows by taking $K = K_{\varepsilon}$.
\end{proof}

Under \Cref{assump:start_time_distortion}, the square-root schedule has the same overall wall-clock scaling as a fixed-threshold \ringmaster run with threshold $R \asymp \sqrt{K}$.
\begin{proposition}[Same Wall-Clock Scaling as Fixed-Threshold \ringmaster]
    \label{prop:same_scaling_ringmaster}
    Assume \Cref{assump:start_time_distortion}.
    Let $R_K \coloneqq \lfloor \sqrt{K} \rfloor$.
    Then
    \begin{equation*}
        T(K)
        \le
        3 C_{\rm time}\,
        R_K\, t(R_K;0).
    \end{equation*}
    Moreover, if \ringmaster is run with the fixed threshold $R \equiv R_K$, then
    \begin{equation*}
        T^{\rm Ringmaster}(K)
        \le
        C_{\rm time}\left\lceil \frac{K}{R_K} \right\rceil t(R_K;0)
        =
        \mathcal{O}\left(
            C_{\rm time}\sqrt{K}\, t(\sqrt{K};0)
        \right).
    \end{equation*}
    Therefore, both methods have the same wall-clock scaling
    \begin{equation*}
        \mathcal{O}\left(
            \sqrt{K}\, t(\sqrt{K};0)
        \right)
    \end{equation*}
    up to constant factors.
\end{proposition}

\begin{proof}
    By \Cref{lem:simplified_formula_start_time},
    \begin{equation*}
        T(K)
        \le
        3 C_{\rm time}\sum_{r=1}^{R_K} t(r;0).
    \end{equation*}
    Since $t(r;0)$ is nondecreasing in $r$, we obtain
    \begin{equation*}
        \sum_{r=1}^{R_K} t(r;0)
        \le
        R_K\, t(R_K;0),
    \end{equation*}
    and hence
    \begin{equation*}
        T(K)
        \le
        3 C_{\rm time}\,
        R_K\, t(R_K;0).
    \end{equation*}

    For fixed-threshold \ringmaster with $R \equiv R_K$, partition the first $K$ iterations into
    \begin{equation*}
        \left\lceil \frac{K}{R_K} \right\rceil
    \end{equation*}
    consecutive blocks.
    Under \Cref{assump:start_time_distortion}, each such block takes at most
    \begin{equation*}
        C_{\rm time}\, t(R_K;0)
    \end{equation*}
    time, so
    \begin{equation*}
        T^{\rm Ringmaster}(K)
        \le
        C_{\rm time}\left\lceil \frac{K}{R_K} \right\rceil t(R_K;0)
        =
        \mathcal{O}\left(
            C_{\rm time}\sqrt{K}\, t(\sqrt{K};0)
        \right).
    \end{equation*}
    This proves the claim.
\end{proof}

The previous proposition shows that, under \Cref{assump:start_time_distortion}, the wall-clock conversion from iterations to time is essentially the same for our method and for fixed-threshold \ringmaster.
Any remaining difference between the final $\varepsilon$-time bounds then comes from the corresponding iteration-complexity estimates, not from the universal computation model itself.

\paragraph{Comparison without additional assumptions.}
For \ringmaster, the iteration complexity in \citet{maranjyan2025ringmaster} is
\begin{equation*}
    \bar{K} \eqdef \left\lceil\frac{48 L \Delta \sigma^2}{\varepsilon^4}\right\rceil .
\end{equation*}
Take the fixed threshold
\begin{equation*}
    R \eqdef \left\lceil \frac{\sigma}{\varepsilon^2} \right\rceil .
\end{equation*}
Applying the fixed-threshold bound above with this choice of $R$ gives
\begin{equation*}
    T_{\varepsilon}^{\rm Ringmaster}
    \le
    T_{\lceil \bar{K}/R \rceil}.
\end{equation*}

For our method, the iteration complexity from \Cref{thm:iteration_complexity} is
\begin{equation*}
    K_{\varepsilon} = \widetilde{\mathcal{O}}\left(\frac{\Psi^4}{\varepsilon^4}\right),
\end{equation*}
and \Cref{thm:time_arbitrary} yields the wall-clock bound
\begin{equation*}
    T_{\varepsilon}
    \le
    S_{3\lfloor \sqrt{K_{\varepsilon}} \rfloor}
    =
    S_{\widetilde{\mathcal{O}}(\Psi^2/\varepsilon^2)} .
\end{equation*}
Therefore, in the arbitrary-time model the direct time comparison is between the two recursions
\begin{equation*}
    T_{\lceil \bar{K}/R \rceil}
    \qquad \text{and} \qquad
    S_{3\lfloor \sqrt{K_{\varepsilon}} \rfloor}.
\end{equation*}
For \ringmaster, the number of recursion steps is
\begin{equation*}
    N_{\rm Ringmaster} \coloneqq \left\lceil \frac{\bar{K}}{R} \right\rceil
    =
    \mathcal{O}\left(\frac{\sigma L\Delta}{\varepsilon^2}\right).
\end{equation*}
For our method, the number of recursion steps is
\begin{equation*}
    N_{\rm ours} \coloneqq 3\lfloor \sqrt{K_{\varepsilon}} \rfloor
    =
    \widetilde{\mathcal{O}}\left(\frac{\Psi^2}{\varepsilon^2}\right)
\end{equation*}
time blocks.
So both methods use on the order of $1/\varepsilon^2$ blocks, up to logarithmic factors.

The difference is in how these blocks are formed.
\ringmaster uses the same threshold $R$ in every block, while our method uses the increasing thresholds
\begin{equation*}
    1,1,1,2,2,2,\dots,\lfloor \sqrt{K_{\varepsilon}} \rfloor,\lfloor \sqrt{K_{\varepsilon}} \rfloor,\lfloor \sqrt{K_{\varepsilon}} \rfloor .
\end{equation*}
Because the quantity $t(r;T)$ depends on both the threshold $r$ and the starting time $T$, the universal computation model does not by itself give a constant-factor comparison between
\begin{equation*}
    T_{N_{\rm Ringmaster}}
    \qquad \text{and} \qquad
    S_{N_{\rm ours}}.
\end{equation*}
Indeed, the worker speeds may change arbitrarily with time, so two recursions with a similar number of steps need not be within a constant factor of each other.

Therefore, in the general case, the clean comparison is that both methods need about $1/\varepsilon^2$ time blocks, but we cannot claim that our bound is equal to the \ringmaster bound up to a universal constant without extra assumptions on the computation-rate functions $\{p_i\}_{i=1}^n$.
